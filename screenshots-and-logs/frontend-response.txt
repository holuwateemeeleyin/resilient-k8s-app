    <p><span class="status" id="status">Checking backend...</span></p>
    <h2>Backend response</h2>
    <pre id="response">Waiting for the API...</pre>
    <button id="refresh">Check backend again</button>
  </main>

  <script>
    async function checkBackend() {
      const status = document.getElementById('status');
      const response = document.getElementById('response');

      status.textContent = 'Checking backend...';

      try {
        const result = await fetch('/api');
        if (!result.ok) {
          throw new Error(`HTTP ${result.status}`);
        }

        const data = await result.json();
        status.textContent = 'Backend connected';
        response.textContent = JSON.stringify(data, null, 2);
      } catch (error) {
        status.textContent = 'Backend unavailable';
        response.textContent = error.message;
      }
    }

    document.getElementById('refresh').addEventListener('click', checkBackend);
    checkBackend();
  </script>
